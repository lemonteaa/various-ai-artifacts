### The Philosophy
1.  **Single Binary**: The API, Scheduler, and Controller run as concurrent processes within one binary.
2.  **Store**: In-memory map (for ultimate speed and minimalism, though data is lost on restart).
3.  **Networking**: Manual Linux networking. We will create a network namespace for every Pod using `ip netns`, set up a `veth` pair, and plug it into a bridge. We tell `podman` to join this pre-existing namespace.
4.  **Ingress**: Handled via simple `iptables` `HostPort` mapping managed by the controller.

```go
// --- Data Models ---

type ContainerPort struct {
	ContainerPort int `yaml:"containerPort"`
	HostPort      int `yaml:"hostPort"`
}

type Container struct {
	Name  string           `yaml:"name"`
	Image string           `yaml:"image"`
	Ports []ContainerPort  `yaml:"ports"`
}

type PodSpec struct {
	Containers []Container `yaml:"containers"`
}

type Pod struct {
	APIVersion string   `yaml:"apiVersion"`
	Kind       string   `yaml:"kind"`
	Metadata   struct {
		Name string `yaml:"name"`
	} `yaml:"metadata"`
	Spec PodSpec `yaml:"spec"`
	
	// Runtime State (Internal)
	Status    string // Pending, Running, Failed
	PodIP     string
	Namespace string
	NodeName  string
}

var (
	store      = make(map[string]Pod)
	storeMutex sync.RWMutex
	bridgeName = "mynet"
	subnet     = "10.244.0."
	nodeName   = "micro-k-node"
)

func main() {
	// Start the sub-systems
	go Scheduler()
	go Controller()

	// Start the API Server
	http.HandleFunc("/apply", applyHandler)
	http.HandleFunc("/pods", listHandler)
	
	fmt.Println("Micro-K API Server listening on :8080")

// --- API Server ---

func applyHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
		return
	}

	var pod Pod
	decoder := yaml.NewDecoder(r.Body)
	if err := decoder.Decode(&pod); err != nil {
		http.Error(w, err.Error(), http.StatusBadRequest)
		return
	}

	// Initialize Pod
	storeMutex.Lock()
	pod.Status = "Pending"
	pod.NodeName = ""
	store[pod.Metadata.Name] = pod
	storeMutex.Unlock()

	w.WriteHeader(http.StatusCreated)
	fmt.Fprintf(w, "Pod %s created\n", pod.Metadata.Name)
}

func listHandler(w http.ResponseWriter, r *http.Request) {
	storeMutex.RLock()
	defer storeMutex.RUnlock()
	
	pods := make([]Pod, 0, len(store))
	for _, p := range store {
		pods = append(pods, p)
	}
	
	json.NewEncoder(w).Encode(pods)
}

// --- Scheduler ---

func Scheduler() {
	ticker := time.NewTicker(5 * time.Second)
	for range ticker.C {
		storeMutex.Lock()
		for name, pod := range store {
			if pod.Status == "Pending" && pod.NodeName == "" {
				pod.NodeName = nodeName
				store[name] = pod
				log.Printf("Scheduled: %s -> %s", name, nodeName)
			}
		}
		storeMutex.Unlock()
	}
}

// --- Controller (The Hacker Magic) ---

func Controller() {
	ticker := time.NewTicker(5 * time.Second)
	for range ticker.C {
		storeMutex.Lock()
		for name, pod := range store {
			// Only handle pods assigned to this node
			if pod.NodeName != nodeName {
				continue
			}

			// Step 1: Create Networking (Namespace + Veth)
			if pod.Status == "Pending" && pod.Namespace == "" {
				pod.Namespace = pod.Metadata.Name
				ip := allocateIP(name)
				pod.PodIP = ip
				
				if err := setupNetworking(pod.Metadata.Name, ip); err != nil {
					log.Printf("Err networking %s: %v", name, err)
					pod.Status = "Failed"
				} else {
					store[name] = pod // Save IP and NS name
				}
			}

			// Step 2: Start Containers (if network ready and containers not running)
			if pod.Status != "Failed" && pod.Status != "Running" && pod.Namespace != "" {
				if err := runContainers(pod); err != nil {
					log.Printf("Err running %s: %v", name, err)
					pod.Status = "Failed"
				} else {
					pod.Status = "Running"
					setupIngress(pod) // Setup HostPort forwarding
					log.Printf("Running: %s (IP: %s)", name, pod.PodIP)
				}
				store[name] = pod
			}
		}
		storeMutex.Unlock()
	}
}

// --- Networking & Runtime (Linux Tooling) ---

func allocateIP(name string) string {
	// Simple deterministic IP allocation based on string hash (for demo)
	// In reality, you'd track assigned IPs
	return subnet + "2" // Hardcoded for simplicity of the demo
}

func setupNetworking(podName, podIP string) error {
	// 1. Create Network Namespace
	exec.Command("ip", "netns", "add", podName).Run()
	
	// 2. Create Veth pair
	vethHost := "veth" + podName[:5] + "0"
	vethPod := "eth0"
	exec.Command("ip", "link", "add", vethHost, "type", "veth", "peer", "name", vethPod).Run()
	
	// 3. Move Pod end into Namespace
	exec.Command("ip", "link", "set", vethPod, "netns", podName).Run()
	
	// 4. Connect Host end to Bridge
	exec.Command("ip", "link", "set", vethHost, "master", bridgeName).Run()
	exec.Command("ip", "link", "set", vethHost, "up").Run()
	
	// 5. Configure inside Namespace
	exec.Command("ip", "netns", "exec", podName, "ip", "link", "set", "lo", "up").Run()
	exec.Command("ip", "netns", "exec", podName, "ip", "addr", "add", podIP+"/24", "dev", vethPod).Run()
	exec.Command("ip", "netns", "exec", podName, "ip", "link", "set", vethPod, "up").Run()
	exec.Command("ip", "netns", "exec", podName, "ip", "route", "add", "default", "via", "10.244.0.1").Run()

	return nil
}

func runContainers(pod Pod) error {
	// We tell Podman to join the specific network namespace we created
	// using the path format for network namespaces
	nsPath := "/var/run/netns/" + pod.Namespace

	for _, c := range pod.Spec.Containers {
		// Check if container already exists (simple restart logic)
		_, err := exec.Command("podman", "inspect", "-f", "{{.State.Running}}", pod.Metadata.Name+"-"+c.Name).CombinedOutput()
		if err == nil {
			continue // Already running
		}

		// Run Podman
		// --network=none is crucial because we handle the networking manually via the namespace join
		cmd := exec.Command("podman", "run", "--name", pod.Metadata.Name+"-"+c.Name, 
			"--network=ns:"+nsPath, 
			"-d", 
			c.Image)
		
		if out, err := cmd.CombinedOutput(); err != nil {
			return fmt.Errorf("podman failed: %v, %s", err, string(out))
		}
	}
	return nil
}

// --- Ingress (Hacker Style) ---

func setupIngress(pod Pod) {
	for _, c := range pod.Spec.Containers {
		for _, p := range c.Ports {
			if p.HostPort > 0 {
				// Use iptables to DNAT HostPort to PodIP:ContainerPort
				// This mimics a NodePort service
				exec.Command("iptables", "-t", "nat", "-C", "PREROUTING", "-p", "tcp", "--dport", fmt.Sprintf("%d", p.HostPort), "-j", "DNAT", "--to-destination", fmt.Sprintf("%s:%d", pod.PodIP, p.ContainerPort)).Run()
				
				// If rule doesn't exist (returned error), add it
				if err := exec.Command("iptables", "-t", "nat", "-A", "PREROUTING", "-p", "tcp", "--dport", fmt.Sprintf("%d", p.HostPort), "-j", "DNAT", "--to-destination", fmt.Sprintf("%s:%d", pod.PodIP, p.ContainerPort)).Run(); err == nil {
					log.Printf("Ingress: Mapped host port %d -> %s:%d", p.HostPort, pod.PodIP, p.ContainerPort)
				}
			}
		}
	}
}
```

### Addressing the Design Decisions

#### 1. The Network Model (CNI-Lite)
We didn't use a full CNI plugin. Instead, the controller acts as the CNI plugin.
*   **Namespace**: Created via `ip netns add`.
*   **Veth**: Created via `ip link add ... type veth`.
*   **Bridge**: The host end of the veth is plugged into `mynet`.
*   **Podman Integration**: Crucially, we run `podman run --network=ns:/var/run/netns/<pod_name>`. This tells Podman **not** to setup its own networking, but to join the network namespace we manually engineered. This satisfies the requirement to use `podman + crun` while retaining full control over the network stack.

#### 2. The "Ingress" Question
In a minimal viable subset (MVP), a full Ingress Controller (like Nginx Ingress watching Ingress resources) is overkill and adds massive complexity (TLS certs, path rewriting, CRDs).

**The Hacker Answer:**
We implement **HostPort** mapping directly in the controller using `iptables`.
When a pod defines a `hostPort`, the controller executes an `iptables` DNAT rule:
`iptables -t nat -A PREROUTING -p tcp --dport 8080 -j DNAT --to-destination 10.244.0.2:80`

This provides immediate, standard access to the services running on the edge node. It is robust, requires no extra components, and works perfectly for an edge scenario where the node is the entry point.

```go
func EnhancedScheduler() {
    ticker := time.NewTicker(5 * time.Second)
    for range ticker.C {
        storeMutex.Lock()
        
        // Filter pending pods
        pendingPods := []string{}
        for name, pod := range store {
            if pod.Status == "Pending" && pod.NodeName == "" {
                pendingPods = append(pendingPods, name)
            }
        }
        
        // Score nodes for each pod
        for _, podName := range pendingPods {
            pod := store[podName]
            bestNode := selectBestNode(pod)
            
            if bestNode != "" {
                pod.NodeName = bestNode
                pod.Status = "Scheduled"
                store[podName] = pod
                log.Printf("Scheduled %s -> %s", podName, bestNode)
            }
        }
        
        storeMutex.Unlock()
    }
}

func selectBestNode(pod Pod) string {
    var bestNode string
    bestScore := -1
    
    for nodeName, node := range clusterState.Nodes {
        if node.Status != "Ready" {
            continue
        }
        
        score := scoreNode(node, pod)
        if score > bestScore {
            bestScore = score
            bestNode = nodeName
        }
    }
    
    return bestNode
}

func scoreNode(node Node, pod Pod) int {
    score := 100 // Base score
    
    // Resource availability
    if node.Used.Pods >= node.Capacity.Pods {
        return -1 // Cannot schedule
    }
    
    // Least resource allocation preferred
    cpuRatio := float64(node.Used.CPU) / float64(node.Capacity.CPU)
    score -= int(cpuRatio * 50)
    
    // Spread pods across nodes
    podCount := countPodsOnNode(node.Name)
    score -= podCount * 10
    
    return score
}
```

### Controller Logic (Enhanced)

```go
func EnhancedController() {
    ticker := time.NewTicker(5 * time.Second)
    for range ticker.C {
        storeMutex.Lock()
        
        for name, pod := range store {
            // Skip pods not assigned to this node
            if pod.NodeName != nodeName {
                continue
            }
            
            // Handle different states
            switch pod.Status {
            case "Scheduled":
                if err := setupNetworking(pod); err == nil {
                    pod.Status = "NetworkReady"
                    store[name] = pod
                }
            case "NetworkReady":
                if err := runContainers(pod); err == nil {
                    pod.Status = "Running"
                    store[name] = pod
                }
            case "Running":
                // Health check
                if !checkContainerHealth(pod) {
                    pod.Status = "Failed"
                    store[name] = pod
                }
            case "Terminating":
                cleanupPod(pod)
                delete(store, name)
            }
        }
        
        storeMutex.Unlock()
    }
}
```

```go
func checkContainerHealth(pod Pod) bool {
    for _, c := range pod.Spec.Containers {
        containerName := pod.Metadata.Name + "-" + c.Name
        
        // Check container status
        cmd := exec.Command("podman", "inspect", "-f", "{{.State.Running}}", containerName)
        output, err := cmd.CombinedOutput()
        
        if err != nil || string(output) != "true\n" {
            log.Printf("Container %s is not running", containerName)
            return false
        }
    }
    return true
}

func getDetailedContainerStatus(pod Pod) map[string]string {
    status := make(map[string]string)
    
    for _, c := range pod.Spec.Containers {
        containerName := pod.Metadata.Name + "-" + c.Name
        
        // Get detailed status
        cmd := exec.Command("podman", "inspect", 
            "-f", "{{.State.Status}}|{{.State.Running}}|{{.State.Restarting}}",
            containerName)
        output, err := cmd.CombinedOutput()
        
        if err != nil {
            status[containerName] = "Error: " + err.Error()
            continue
        }
        
        parts := strings.Split(strings.TrimSpace(string(output)), "|")
        if len(parts) == 3 {
            status[containerName] = fmt.Sprintf("Status: %s, Running: %s, Restarting: %s",
                parts[0], parts[1], parts[2])
        }
    }
    
    return status
}
```

## Key Architectural Insights

### Scheduler vs Controller Responsibilities:
- **Scheduler**: "Where should this run?" - Placement decisions based on resources, affinity, etc.
- **Controller**: "Is it running correctly?" - Ensures desired state matches actual state

### Multi-Node Challenges Solved:
1. **Network Flattening**: VXLAN overlay network provides connectivity across nodes
2. **Service Discovery**: Use CoreDNS or simple DNS round-robin
3. **High Availability**: Multiple control plane nodes with leader election
4. **Data Consistency**: Upgrade to etcd for distributed storage

### 1. DNS Server Implementation (Add to main.go)

```go

import (
    "github.com/miekg/dns"
)

// DNS Server - runs on control plane
func RunDNSServer() {
    server := &dns.Server{
        Addr:    ":53", // Or bind to bridge IP: "10.244.0.1:53"
        Net:     "udp",
        Handler: dnsHandler{},
    }
    
    log.Printf("Micro-K DNS Server starting on :53")
    go func() {
        if err := server.ListenAndServe(); err != nil {
            log.Fatalf("Failed to start DNS server: %v", err)
        }
    }()
}

type dnsHandler struct{}

func (h dnsHandler) ServeDNS(w dns.ResponseWriter, r *dns.Msg) {
    msg := new(dns.Msg)
    msg.SetReply(r)
    msg.Authoritative = true
    
    for _, question := range r.Question {
        if question.Qtype == dns.TypeA {
            // Extract pod name from query
            // Assumes queries like "my-nginx.pod.local" or "my-nginx"
            podName := strings.TrimSuffix(question.Name, ".pod.local.")
            podName = strings.TrimSuffix(podName, ".")
            
            storeMutex.RLock()
            pod, exists := store[podName]
            storeMutex.RUnlock()
            
            if exists && pod.PodIP != "" {
                rr, err := dns.NewRR(fmt.Sprintf("%s 5 IN A %s", question.Name, pod.PodIP))
                if err == nil {
                    msg.Answer = append(msg.Answer, rr)
                    log.Printf("DNS: %s -> %s", podName, pod.PodIP)
                }
            } else {
                log.Printf("DNS: Pod %s not found", podName)
                msg.SetRcode(r, dns.RcodeNameError)
            }
        }
    }
    
    w.WriteMsg(msg)
}
```

```go
func Controller() {
    ticker := time.NewTicker(5 * time.Second)
    for range ticker.C {
        storeMutex.Lock()
        
        // 1. 处理Pod（原有逻辑）
        for name, pod := range store {
            if pod.NodeName != nodeName {
                continue
            }
            // ... 现有Pod处理逻辑 ...
        }
        
        // 2. 处理Service（新增）
        reconcileServices()
        
        storeMutex.Unlock()
    }
}

func reconcileServices() {
    for svcName, svc := range svcStore {
        if svc.Status == "Active" {
            // 检查后端Pod是否仍然存在
            if !validateServiceBackends(svc) {
                // 后端变化，需要重建iptables规则
                cleanupServiceRules(svc)
                setupServiceRules(svc)
            }
        } else {
            // 新Service，设置iptables规则
            if err := setupServiceRules(svc); err == nil {
                svc.Status = "Active"
                svcStore[svcName] = svc
                log.Printf("Service %s activated", svcName)
            }
        }
    }
}

// 检查Service后端是否有效
func validateServiceBackends(svc Service) bool {
    for _, backend := range svc.Spec.Backends {
        if pod, exists := store[backend.PodName]; !exists || pod.Status != "Running" {
            return false
        }
    }
    return true
}
```

### 第四步：核心 - iptables规则设置

```go
// setupServiceRules 创建Service的所有iptables规则
func setupServiceRules(svc Service) error {
    vip := svc.Spec.ClusterIP
    if vip == "" {
        vip = allocateServiceIP()
        svc.Spec.ClusterIP = vip
        svcStore[svc.Metadata.Name] = svc
    }
    
    for _, port := range svc.Spec.Ports {
        // 1. 创建KUBE-SERVICES链中的规则（目标到Service链）
        rule := fmt.Sprintf("-t nat -A KUBE-SERVICES -d %s/32 -p %s --dport %d -j KUBE-SVC-%s",
            vip, port.Protocol, port.Port, svc.Metadata.Name)
        exec.Command("sh", "-c", "iptables "+rule).Run()
        
        // 2. 创建Service特定链（KUBE-SVC-<name>）
        createServiceChain(svc, port)
        
        // 3. 创建到各个后端的规则
        for i, backend := range svc.Spec.Backends {
            pod, exists := store[backend.PodName]
            if !exists {
                log.Printf("Warning: Backend pod %s not found", backend.PodName)
                continue
            }
            
            // 使用random模式实现简单的负载均衡
            // 1/3的概率匹配（如果有3个后端）
            probability := fmt.Sprintf("0.%03d", 1000/len(svc.Spec.Backends))
            
            rule := fmt.Sprintf("-t nat -A KUBE-SVC-%s -m statistic --mode random --probability %s -j DNAT --to-destination %s:%d",
                svc.Metadata.Name, probability, pod.PodIP, backend.Port)
            exec.Command("sh", "-c", "iptables "+rule).Run()
        }
        
        // 4. 如果是NodePort类型，添加PREROUTING规则
        if svc.Spec.Type == ServiceTypeNodePort {
            // 简化：假设NodePort映射到Service Port
            nodePort := port.Port + 30000 // 偏移量
            rule := fmt.Sprintf("-t nat -A PREROUTING -p %s --dport %d -j KUBE-SVC-%s",
                port.Protocol, nodePort, svc.Metadata.Name)
            exec.Command("sh", "-c", "iptables "+rule).Run()
        }
    }
    
    return nil
}

func createServiceChain(svc Service, port ServicePort) {
    // 清除旧链（如果存在）
    exec.Command("sh", "-c", "iptables -t nat -F KUBE-SVC-"+svc.Metadata.Name).Run()
    exec.Command("sh", "-c", "iptables -t nat -X KUBE-SVC-"+svc.Metadata.Name).Run()
    
    // 创建新链
    exec.Command("sh", "-c", "iptables -t nat -N KUBE-SVC-"+svc.Metadata.Name).Run()
}

func cleanupServiceRules(svc Service) {
    // 删除Service链
    exec.Command("sh", "-c", "iptables -t nat -F KUBE-SVC-"+svc.Metadata.Name).Run()
    exec.Command("sh", "-c", "iptables -t nat -X KUBE-SVC-"+svc.Metadata.Name).Run()
    
    // 从KUBE-SERVICES中删除引用
    // （简化版：完整实现需要精确匹配和删除）
}
```

## 🎯 回答你的核心问题

### Q1: iptables需要处理返回包吗？

**答：不需要！** conntrack自动处理。

```bash
# 观察conntrack如何追踪连接
sudo conntrack -E -p tcp

# 在另一个终端：
sudo ip netns exec micro-pod-1 curl http://10.96.0.2

# 你会看到类似这样的输出：
# [NEW] tcp=6 src=10.244.0.1 dst=10.96.0.2 sport=xxxxx dport=80 [UNREPLIED]
# [UPDATE] tcp=6 src=10.244.0.1 dst=10.96.0.2 sport=xxxxx dport=80 ...
# [UPDATE] tcp=6 src=10.244.0.2 dst=10.244.0.1 sport=8080 dport=xxxxx ...

# 注意返回包的src已经被改回了VIP（10.96.0.2）
```

### Q2: 负载均衡只在有Pod的节点间吗？

**答：负载均衡在Pod级别，不依赖节点！**

```
Node A:
  - Pod 1 (10.244.0.2)
  - Pod 2 (10.244.0.3)

Node B:
  - Pod 3 (10.244.0.4)

Service (10.96.0.2:80) 负载均衡到：
  - 10.244.0.2:8080 (Node A)
  - 10.244.0.3:8080 (Node A)  
  - 10.244.0.4:8080 (Node B)
```

**关键洞察：**
- Service不知道也不关心Pod在哪台节点上
- 只要网络是扁平的（跨节点可达），负载均衡就能工作
- 这就是为什么K8s需要CNI网络插件提供跨节点Pod网络

### Q3: 调度算法和亲和性？

**答：是的，Service确实影响调度！**

这就是为什么K8s有：
1. **Pod Anti-Affinity**：确保同一应用的Pod分散到不同节点
2. **TopologySpreadConstraints**：控制Pod跨域分布

```go
// 简化的调度逻辑示例
func scoreNode(node Node, pod Pod) int {
    score := 100
    
    // ... 原有资源评分 ...
    
    // 新增：检查Service相关
    if isPartOfService(pod) {
        svc := getServiceForPod(pod)
        podCountOnNode := countServicePodsOnNode(svc, node.Name)
        
        // 倾向于分散（anti-affinity）
        score -= podCountOnNode * 20
    }
    
    return score
}
```

### Q4: 为什么这叫"编排"？

你说的完全正确！

```
用户需求："我要3个nginx实例，用一个Service暴露"

控制器背后的工作：
┌─────────────────────────────────────────────┐
│ 1. 创建3个Pod资源                            │
│ 2. 调度器选择3个不同节点（考虑anti-affinity） │
│ 3. Controller在节点1启动Pod1                 │
│ 4. Controller在节点2启动Pod2                 │
│ 5. Controller在节点3启动Pod3                 │
│ 6. Controller分配Service VIP                 │
│ 7. Controller在所有节点设置iptables规则       │
│ 8. 当Pod崩溃时，Controller自动重启            │
│ 9. 当Pod迁移时，自动更新iptables规则          │
└─────────────────────────────────────────────┘

用户看到的："搞定，Service可以用了"
```
